cmake_minimum_required(VERSION 3.15)
project(orangeArray LANGUAGES CXX)

if(NOT MAYA_LOCATION)
    message(FATAL_ERROR "Set MAYA_LOCATION to the Maya installation, e.g. C:/Program Files/Autodesk/Maya2027")
endif()

add_library(orangeArray MODULE orangeArray.cpp)
target_include_directories(orangeArray PRIVATE "${MAYA_LOCATION}/include")
target_link_directories(orangeArray PRIVATE "${MAYA_LOCATION}/lib")
target_link_libraries(orangeArray PRIVATE OpenMaya Foundation)
target_compile_features(orangeArray PRIVATE cxx_std_14)
target_compile_definitions(orangeArray PRIVATE NT_PLUGIN REQUIRE_IOSTREAM)
set_target_properties(orangeArray PROPERTIES PREFIX "" SUFFIX ".mll" WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
    set_property(TARGET orangeArray PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    target_compile_options(orangeArray PRIVATE /W4 /permissive-)
    target_link_options(orangeArray PRIVATE /EXPORT:initializePlugin /EXPORT:uninitializePlugin)
endif()
